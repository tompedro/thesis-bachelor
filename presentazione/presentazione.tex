	% ============ 1. IL PROBLEMA E LA STRATEGIA ============
	\begin{frame}{Il problema e la strategia}
		\begin{itemize}
			\item Meccanica classica e  Meccanica quantistica diverso formalismo ma stesse strutture. 
			\item Esiste un linguaggio più primitivo, comune a entrambe le teorie?
			\item Strategia Algebrica: \textbf{C*-algebre}.
		\end{itemize}
		
		\begin{block}{C*-algebra}
			Una \emph{C*-algebra} è un'algebra di Banach $\mathfrak{A}$ su $\mathbb{C}$ munita di un'involuzione $*:\mathfrak{A} \to \mathfrak{A}$ che soddisfa, per ogni $A, B \in \mathfrak{A}$ e $\lambda \in \mathbb{C}$
			\begin{itemize}
				\item $(A^*)^* = A$, $(A+B)^* = A^* + B^*$ e $(\lambda A)^* = \overline{\lambda}\, A^*$
				\item $(AB)^* = B^* A^*$
			\end{itemize}
			tale che la norma soddisfi la \emph{C*-identità}
			\[
			\|A^* A\| = \|A\|^2, \qquad \forall A \in \mathfrak{A}.
			\]
		\end{block}
	\end{frame}
	
	% ============ CAPITOLO 1 ============
	
	% ---- 2. Spazio delle fasi ----
	\begin{frame}{Lo spazio delle fasi}
		\begin{itemize}
			\item Lo spazio delle fasi: spazio liscio dove i punti sono gli stati del sistema
			
			\item L'energia $H$ genera la dinamica tramite la forma simplettica
			\[
			\omega(X_H,\,\cdot\,) = dH.
			\]
			\item Le traiettorie del sistema sono le curve determinate dal campo vettoriale $X_H$.
		\end{itemize}
		
		\begin{block}{Esempio: $\mathbb{R}^2$, $(q,p)$}
			\[
			\omega = dq \wedge dp, \qquad J = \begin{pmatrix} 0 & 1 \\ -1 & 0 \end{pmatrix}
			\]
			\[
			X_H = J\nabla H \qquad \Longrightarrow \qquad \dot q = \frac{\partial H}{\partial p}, \quad \dot p = -\frac{\partial H}{\partial q}
			\]
		\end{block}
	\end{frame}
	
	% ---- 3. Parentesi di Poisson ----
	\begin{frame}{Le parentesi di Poisson}
		\begin{itemize}
			\item Parentesi di Poisson: $\omega$ su campi vettoriali di due osservabili $f,g$
			\[
			\{f,g\} = \omega(X_f,X_g), \qquad \dot f = \{f,H\}.
			\]
		\end{itemize}
		\begin{block}{Equazioni di Hamilton tramite Parentesi di Poisson}
			\centering
			\[
			\dot q^i = \{q^i,H\} = \frac{\partial H}{\partial p_i}, \qquad \dot p_i = \{p_i,H\} = -\frac{\partial H}{\partial q^i}
			\]
		\end{block}
		\begin{itemize}
			\item Gli osservabili: $C^\infty(M;\mathbb{R})$ possiamo immergerli in $C^*$-algebra $C^0(M; \mathbb{C})$.
		\end{itemize}
	\end{frame}
	
	% ---- 4. Gelfand-Naimark, con riquadro ----
	\begin{frame}{Il teorema di Gelfand--Naimark}
	\begin{itemize}
		\item Spettro $X = \sigma(\mathfrak{A})$ : spazio degli stati puri del sistema
	\end{itemize}
	\begin{block}{Il teorema di Gelfand--Naimark}
		\centering
		\[
		C^*\text{-algebra commutativa } \mathfrak{A} \qquad\Longrightarrow\qquad \mathfrak{A} \;\cong\; C_0\big(X\big)
		\]
	\end{block}
	\begin{itemize}
		\item Partendo dall'algebra astratta si ottiene $X$ 
	\end{itemize}
\end{frame}
	
	% ---- 5. Il limite della costruzione ----
	\begin{frame}{Un ingrediente esterno}
		\begin{itemize}
			\item $X$ è solo uno spazio topologico: nessuna derivata e nessuna parentesi di Poisson.
			\item Serve scegliere una sottoalgebra densa di funzioni lisce.
		\end{itemize}
		\vspace{0.4cm}
		\alert{La parentesi di Poisson non è intrinseca all'algebra ma va importata dall'esterno.}
	\end{frame}

	% ---- 6. Costruzione dell'algebra degli osservabili, a riquadretti ----
	\begin{frame}{La costruzione dell'algebra degli osservabili}
		Punto di vista operativo: $C^*$-algebra da un set minimale.
		
		\vspace{0.25cm}
		\begin{columns}[T,onlytextwidth]
			\column{0.48\textwidth}
			\begin{block}{1. Potenze e Somme}
				\centering\footnotesize
				\vspace{0.15cm}
				$\quad A^n, \quad A^0 \equiv 1, \quad A+B$
				\vspace{0.15cm}
			\end{block}
			\vspace{0.2cm}
			\begin{block}{3. Immersione in $C^*$-algebra}
				\centering\footnotesize
				\vspace{0.1cm}
				$A \circ B = \dfrac12 (AB + BA)$
			\end{block}
			
			\column{0.48\textwidth}
			\begin{block}{2. Prodotto di Jordan}
				\centering\footnotesize
				\vspace{0.1cm}
				$A \circ B := \dfrac{1}{2}\big((A{+}B)^2 {-} A^2 {-} B^2\big)$
			\end{block}
			\vspace{0.2cm}
			\begin{block}{4. Parentesi di Poisson}
				\centering\footnotesize
				\vspace{0.1cm}
				$\{A,B\} := \dfrac{1}{i\hbar}[A,B]$
			\end{block}
		\end{columns}
		
	\end{frame}
	
	% ---- 7. GNS: risultato centrale ----
	\begin{frame}{La costruzione GNS}
		Stato: un funzionale lineare che restituisce il valore di aspettazione
		$$\omega(A) = \langle A \rangle, \qquad A \in \mathfrak{A}.$$
		\begin{block}{Teorema Gelfand--Naimark--Segal}
			Ogni stato $\omega$ su una C*-algebra $\mathfrak{A}$ genera uno spazio di Hilbert $\mathcal H_\omega$, una rappresentazione $\pi_\omega : \mathfrak{A} \to \mathcal{B}(\mathcal{H}_\omega)$ e un vettore $\Psi_\omega \in \mathcal{H}_\omega$ tali che
			\[
			\omega(A) = \big\langle \Psi_\omega,\, \pi_\omega(A)\,\Psi_\omega \big\rangle \qquad \forall A \in \mathfrak{A}.
			\]
		\end{block}
		I postulati di Dirac--von Neumann diventano giustificati.
	\end{frame}
	
	% ============ CAPITOLO 3 ============
	
	% ---- 8. Spazio di Hilbert proiettivo ----
	\begin{frame}{Lo spazio di Hilbert proiettivo}
		\begin{itemize}
			\item Gli stati fisici di un sistema quantistico sono definiti a meno di fase e di norma.
			
			\item $\mathbb P\mathcal H$ è uno spazio liscio come in meccanica classica.
			
			\item Gli osservabili  sono mappe $f_A: \mathbb{P} \mathcal{H} \to \mathbb{R}$ tali che $f_A = \langle A \rangle$ con $A$ osservabile.
		\end{itemize}
	\end{frame}
	
	% ---- 9. Struttura di Kähler + hbar come curvatura ----
	\begin{frame}{La struttura di Kähler}
		
		\textbf{Struttura di Kähler}: metrica riemanniana $g$ e una forma simplettica $\omega$. 
		$$
			\langle \cdot, \cdot \rangle = g(\cdot, \cdot) + i \omega(\cdot, \cdot)
		$$
		\textbf{Teorema di rigidità}: $\mathbb P\mathcal H$ ha una curvatura costante $1/\hbar$.

		\begin{block}{Hamilton $\iff$ Schr\"odinger}
			\centering
			\[
			\omega\big(X_{\langle H\rangle},\,\cdot\,\big) = d\langle H\rangle \qquad\Longleftrightarrow\qquad i\hbar\,\dot\Psi = H\Psi
			\]
		\end{block}
	\end{frame}
	

	
	% ---- 10. Indeterminazione e regola di Born ----
	\begin{frame}{Indeterminazione e regola di Born}
		\begin{itemize}
			\item La varianza di un'osservabile $f$ si legge dalla metrica $g$
			\item La disuguaglianza di Robertson--Schrödinger è Cauchy--Schwarz per $g$.
		\end{itemize}
		\begin{block}{Indeterminazione su un osservabile}
			\centering
			\[
			(\Delta f)^2 = \frac{\hbar}{2}\, g(X_f,X_f) \qquad (\Delta f)(\Delta k) \geq \frac{\hbar}{2} \sqrt{\omega(X_f, X_k)^2 + g(X_f,X_k)^2}
			\]
		\end{block}
	
		\begin{itemize}
			\item Integrando $g$ si ottiene la probabilità di transizione.
		\end{itemize}
	\end{frame}

\begin{frame}{Esempio: $\mathbb{R}^2$ con coordinate $(q,p)$}
	\begin{exampleblock}{La forma simplettica canonica su $\mathbb{R}^2$}
		Per il Teorema di Darboux, $\omega = dq \wedge dp$, che si rappresenta con la matrice antisimmetrica
		\[
		[\omega] = \begin{pmatrix} 0 & 1 \\ -1 & 0 \end{pmatrix} =: J,
		\]
		ottenuta calcolando $\omega$ sui vettori di base. Dato $H=H(q,p)$, il campo hamiltoniano $X_H = a\,\partial_q + b\,\partial_p$ è definito da $\omega(X_H,\cdot) = dH$. Per antisimmetria,
		\[
		\omega(X_H,\cdot) = a\,dp - b\,dq \;=\; \partial_q H\, dq + \partial_p H\, dp,
		\]
		da cui, confrontando i coefficienti, $a = \partial_p H$, $b = -\partial_q H$, nessuna inversione di matrice, cioè
		\[
		X_H = J\,\nabla H = \begin{pmatrix} \partial H/\partial p \\ -\partial H/\partial q \end{pmatrix}.
		\]
	\end{exampleblock}
\end{frame}


\begin{frame}{Dal prodotto di Jordan alla JB-algebra}
	\begin{block}{Prodotto di Jordan}
		\[
		A \circ B := \tfrac12\big[(A+B)^2 - A^2 - B^2\big] = B \circ A
		\]
	\end{block}
	\begin{itemize}
		\item Commutativo, ma in generale \alert{non associativo}.
		\item Per potenze di un solo osservabile coincide con il prodotto ordinario: $A \circ A^n = A^{n+1}$.
	\end{itemize}
	
	\begin{block}{JB-algebra}
		Spazio di Banach reale $\mathcal O$ con $\circ$ commutativo, distributivo, debolmente associativo
		\[
		A^2 \circ (B \circ A) = (A^2 \circ B) \circ A,
		\]
		tale che $\|A\circ B\|\le \|A\|\,\|B\|$, $\ \|A\|^2 = \|A^2\|$, $\ \|A^2\| \le \|A^2+B^2\|$.
	\end{block}
\end{frame}

\begin{frame}{L'immersione in una C*-algebra}
	\begin{block}{Immersione}
		$\mathcal O$ si immerge in una C*-algebra $\mathfrak{A}$ se esiste un'isometria lineare $\mathcal O \hookrightarrow \mathfrak{A}$ la cui immagine genera $\mathfrak{A}$, tale che per ogni $A,B\in\mathcal O$
		\[
		A^*=A \ \text{in } \mathfrak{A}, \qquad A\circ B = \tfrac12(AB+BA).
		\]
	\end{block}
	\begin{itemize}
		\item $\mathcal O$ si identifica con un sottospazio chiuso di $\mathfrak{A}_{sa}$.
		\item Una JB-algebra che ammette questa immersione si dice \emph{JC-algebra}.
	\end{itemize}
	
	\begin{block}{Complessificazione e prodotto ricostruito}
		\[
		\mathcal O_{\mathbb C} = \{A+iB \mid A,B\in\mathcal O\}, \qquad AB = A\circ B + \tfrac{i\hbar}{2}\{A,B\}.
		\]
	\end{block}
	Il completamento di $\mathcal O_{\mathbb C}$ è la C*-algebra unitale $\mathfrak{A}$: il prodotto associativo si ricostruisce sommando il prodotto di Jordan e la parentesi di Lie.
\end{frame}

\begin{frame}{Il prodotto di Kähler $*_\nu$}
	Gli isomorfismi musicali $I,G$ della struttura di Kähler $(\mathbb{P} \mathcal{H},\omega,g)$ definiscono due parentesi sulle funzioni lisce.
	\begin{block}{Parentesi di Poisson e di Riemann}
		\[
		\{f,h\} := \omega(I df, I dh) \qquad (\!( f,l )\!) := g( Gdf, G\,dh).
		\]
	\end{block}
	\begin{itemize}
		\item $\{\cdot,\cdot\}$ dipende solo da $\omega$: genera la dinamica.
		\item $(\!(\cdot,\cdot)\!)$ dipende solo da $g$: codifica la dispersione statistica.
	\end{itemize}
	\begin{block}{Parentesi e prodotti di Kähler}
		\[
		\langle f,h\rangle := (\!( f,h )\!) + i\{f,h\}, \qquad f *_\nu l := \tfrac{\nu}{2}\langle f,h\rangle + fh.
		\]
	\end{block}
	$*_\nu$ è il candidato geometrico al prodotto associativo tra osservabili, per un parametro libero $\nu\in\mathbb R$.
\end{frame}

\begin{frame}{Perché $\nu=\hbar$}
	\begin{block}{Corollario 3.5.5}
		Se $(\!(\langle A\rangle,\langle B\rangle)\!)$ e $\{\langle A\rangle,\langle B\rangle\}$ non sono identicamente nulle su $\mathbb{P} \mathcal{H}$, allora
		\[
		\langle A\rangle *_\nu \langle B\rangle = \langle AB\rangle \ \text{su } \mathbb{P} \mathcal{H} \iff \nu = \hbar.
		\]
	\end{block}
	\textbf{Idea della dimostrazione.} Poniamo $f=\langle A\rangle$, $k=\langle B\rangle$.
	\begin{itemize}
		\item Algebricamente, $AB = A\circ B + \tfrac{i\hbar}{2}\{A,B\}$.
		\item Geometricamente, $\langle A\circ B\rangle = fk + \tfrac{\hbar}{2}(\!(f,k)\!)$ e $\langle\{A,B\}\rangle = \{f,k\}$.
		\item Confrontando con $f*_\nu k = fk + \tfrac{\nu}{2}\big[(\!(f,k)\!)+i\{f,k\}\big]$,
		\[
		f *_\nu k - \langle AB\rangle = \tfrac{\nu-\hbar}{2}\big[(\!(f,k)\!) + i\{f,k\}\big].
		\]
	\end{itemize}
	Si annulla per ogni stato solo se $\nu=\hbar$: la stessa normalizzazione che riproduce il flusso di Schrödinger fissa anche il prodotto geometrico.
\end{frame}

\begin{frame}{Perché $q$ e $p$ non ammettono una C*-norma}
	\begin{block}{Ostruzione}
		Da $[q,p]=i\hbar$ segue $[p,q^n] = -in\hbar\, q^{n-1}$, e quindi, se esistesse una C*-norma,
		\[
		\|q\|\,\|p\| \ge \tfrac{n\hbar}{2} \qquad \forall n\in\mathbb N.
		\]
	\end{block}
	\begin{itemize}
		\item $\|q\|$ e $\|p\|$ non possono essere entrambe finite.
		\item Teorema di Wintner: per operatori \emph{limitati} $A,B$, $AB-BA \ne c 1$.
	\end{itemize}
	\begin{block}{Soluzione di Weyl}
		Si sostituiscono $q,p$ con gli esponenziali unitari, limitati per costruzione,
		\[
		U(\alpha)=e^{i\alpha q}, \quad V(\beta) = e^{i\beta p}, \qquad U(\alpha)V(\beta) = V(\beta)U(\alpha)\,e^{-i\alpha\beta}.
		\]
	\end{block}
	La C*-algebra di Weyl $\mathfrak{A}_W$ generata da $U,V$ è l'oggetto su cui si applica Gelfand--Naimark non commutativo.
\end{frame}

\begin{frame}{Il teorema di unicità di von Neumann}
	\begin{itemize}
		\item GNS garantisce l'\alert{esistenza} di una rappresentazione di $\mathfrak{A}_W$, non l'unicità.
		\item Una rappresentazione è \emph{regolare} se $\pi(U(\alpha))$, $\pi(V(\beta))$ sono gruppi unitari fortemente continui: condizione necessaria per avere generatori autoaggiunti (teorema di Stone).
	\end{itemize}
	\begin{block}{Teorema di von Neumann}
		Tutte le rappresentazioni regolari e irriducibili di $\mathfrak{A}_W$ sono unitariamente equivalenti.
	\end{block}
	\begin{itemize}
		\item La rappresentazione di Schrödinger su $L^2(\mathbb R^s)$ è, a meno di equivalenza unitaria, \alert{l'unica} rappresentazione fisicamente ammissibile con un numero finito di gradi di libertà.
		\item L'ipotesi di dimensione finita è cruciale: cade in teoria dei campi.
	\end{itemize}
\end{frame}

\begin{frame}{La curvatura di $\mathbb{P} \mathcal{H}$ e il teorema di Hawley--Igusa}
	\begin{block}{Curvatura sezionale olomorfa}
		Per il potenziale di Kähler $K_\lambda = \tfrac{\lambda}{2}\ln(1+\|z\|^2)$, la curvatura sezionale olomorfa di $\mathbb{P} \mathcal{H}$ è costante, pari a $2/\lambda$.
	\end{block}
	Con la normalizzazione fisica $\lambda = 2\hbar$ (Fissata dal flusso di Schrödinger):
	\[
	\text{curvatura di } \mathbb{P} \mathcal{H} = \frac{1}{\hbar}.
	\]
	\begin{block}{Teorema di Hawley--Igusa}
		A meno di isomorfismo, gli spazi proiettivi sono le \alert{uniche} varietà di Kähler connesse, semplicemente connesse, complete, a curvatura sezionale olomorfa costante e positiva.
	\end{block}
\end{frame}

\begin{frame}{Dimensione infinita e settori di superselezione}
	\begin{itemize}
		\item La costruzione GNS resta valida senza modifiche per algebre di campo.
		\item Cade però l'\alert{unicità}: nell'analogo infinito-dimensionale dell'algebra di Weyl, rappresentazioni regolari e irriducibili distinte non sono più unitariamente equivalenti.
	\end{itemize}
	\begin{block}{Conseguenza fisica}
		Temperature di equilibrio diverse, o masse diverse, corrispondono a rappresentazioni \emph{inequivalenti}: situazioni fisicamente distinte, non un'unica immagine di Hilbert.
	\end{block}
	È il regime della teoria dei campi, della meccanica statistica quantistica e dei sistemi a molti corpi, fuori dall'ambito diretto di questa tesi, ma prosecuzione naturale del formalismo algebrico.
\end{frame}

		Ogni stato $\omega$, anche misto, genera comunque uno spazio di Hilbert $\mathcal H_\omega$ con vettore ciclico $\Psi_\omega$:

\end{frame}
